#!/usr/bin/env bash
# Server Pulse setup for Ubuntu 22.04 / 24.04 — no reboot.
# Usage:  bash server-pulse-setup.sh   (it asks for your two passwords)
set -euo pipefail

if [ -z "${OWNER_PASS:-}" ]; then read -rsp "Owner password (to open btop in browser): " OWNER_PASS; echo; fi
if [ -z "${APP_PASS:-}" ]; then read -rsp "Dashboard password (used by the app): " APP_PASS; echo; fi

echo "== Step 1: btop + ttyd =="
sudo apt update && sudo apt install -y btop curl wget
sudo wget -qO /usr/local/bin/ttyd "https://github.com/tsl0922/ttyd/releases/download/1.7.7/ttyd.$(uname -m)"
sudo chmod +x /usr/local/bin/ttyd
sudo tee /etc/systemd/system/btop-web.service > /dev/null <<UNIT
[Unit]
Description=btop live in browser (ttyd)
After=network-online.target
Wants=network-online.target

[Service]
User=$USER
Environment=TERM=xterm-256color
ExecStart=/usr/local/bin/ttyd -p 7681 -i lo -t titleFixed=btop-live -t fontSize=14 -t disableLeaveAlert=true btop
Restart=always
RestartSec=3

[Install]
WantedBy=multi-user.target
UNIT
sudo systemctl daemon-reload && sudo systemctl enable --now btop-web

echo "== Step 2: Netdata =="
wget -qO /tmp/netdata-kickstart.sh https://get.netdata.cloud/kickstart.sh
sh /tmp/netdata-kickstart.sh --stable-channel --disable-telemetry --non-interactive
if ! grep -q "dbengine tier 0 retention time" /etc/netdata/netdata.conf 2>/dev/null; then
  sudo tee -a /etc/netdata/netdata.conf > /dev/null <<'CONF'

[web]
bind to = 127.0.0.1

[db]
mode = dbengine
dbengine tier 0 retention time = 30d
CONF
fi
sudo systemctl restart netdata   # restarts only Netdata, not the system

echo "== Step 3: Caddy =="
sudo apt install -y debian-keyring debian-archive-keyring apt-transport-https gnupg
curl -1sLf 'https://dl.cloudsmith.io/public/caddy/stable/gpg.key' | sudo gpg --dearmor --yes -o /usr/share/keyrings/caddy-stable-archive-keyring.gpg
curl -1sLf 'https://dl.cloudsmith.io/public/caddy/stable/debian.deb.txt' | sudo tee /etc/apt/sources.list.d/caddy-stable.list > /dev/null
sudo apt update && sudo apt install -y caddy
OWNER_HASH=$(caddy hash-password --plaintext "$OWNER_PASS")
APP_HASH=$(caddy hash-password --plaintext "$APP_PASS")
sudo tee /etc/caddy/Caddyfile > /dev/null <<CADDY
:8080 {
	bind 127.0.0.1
	encode zstd gzip
	basic_auth {
		owner ${OWNER_HASH}
		dashboard ${APP_HASH}
	}
	redir /history /history/
	handle_path /history/* {
		reverse_proxy 127.0.0.1:19999
	}
	handle {
		reverse_proxy 127.0.0.1:7681
	}
}
CADDY
caddy validate --config /etc/caddy/Caddyfile --adapter caddyfile
sudo systemctl enable --now caddy && sudo systemctl restart caddy   # restarts only Caddy

echo "== Step 4: Keep awake (no reboot) =="
sudo systemctl mask sleep.target suspend.target hibernate.target hybrid-sleep.target
sudo mkdir -p /etc/systemd/logind.conf.d
sudo tee /etc/systemd/logind.conf.d/server-pulse.conf > /dev/null <<'LOGIN'
[Login]
HandleLidSwitch=ignore
HandleLidSwitchExternalPower=ignore
HandleLidSwitchDocked=ignore
LOGIN
sudo systemctl kill -s HUP systemd-logind   # reloads settings, does not log you out

echo "== Step 5: Tailscale Funnel =="
command -v tailscale >/dev/null || curl -fsSL https://tailscale.com/install.sh | sh
echo ">> Open the login link printed below in your browser."
sudo tailscale up
echo ">> If a link is printed below, open it to enable HTTPS + Funnel."
sudo tailscale funnel --bg 8080

echo
echo "== Checks =="
systemctl is-active btop-web netdata caddy
systemctl is-enabled btop-web netdata caddy tailscaled
curl -s -o /dev/null -w 'Netdata via Caddy: %{http_code} (expect 200)\n' -u "dashboard:$APP_PASS" http://127.0.0.1:8080/history/api/v1/info
tailscale funnel status

cat <<'NEXT'

Done — no reboot needed.
1. In the Tailscale admin console: Machines -> ... -> Disable key expiry.
2. Dashboard link (use your .ts.net host from above, no https:// before it):
   https://dashboard:YOUR_APP_PASS@YOUR-HOST.ts.net/history
   Paste it on the Server Pulse Connect page.
NEXT
